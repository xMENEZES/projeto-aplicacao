60 def _find_path(self, url):

63 if url.startswith('file://localhost/'): 

64 return url[16:]

99 """Delete expired meta-data.""" 

100 if not self.expires: 

101 return 

102 epoch = datetime(1970, 1, 1, tzinfo=self.app.timezone) 

103 now_ts = (self.app.now() - epoch).total_seconds() 

104 cutoff_ts = now_ts - self.expires
