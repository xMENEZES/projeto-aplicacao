1"""The ``celery purge`` program, used to delete messages from queues.""" 

2import click 

3 

4from celery.bin.base import COMMA_SEPARATED_LIST, CeleryCommand, CeleryOption, handle_preload_options 

5from celery.utils import text 

6 

7 

8@click.command(cls=CeleryCommand, context_settings={ 

9 'allow_extra_args': True 

10}) 

11@click.option('-f', 

12 '--force', 

13 cls=CeleryOption, 

14 is_flag=True, 

15 help_group='Purging Options', 

16 help="Don't prompt for verification.") 

17@click.option('-Q', 

18 '--queues', 

19 cls=CeleryOption, 

20 type=COMMA_SEPARATED_LIST, 

21 help_group='Purging Options', 

22 help="Comma separated list of queue names to purge.") 

23@click.option('-X', 

24 '--exclude-queues', 

25 cls=CeleryOption, 

26 type=COMMA_SEPARATED_LIST, 

27 help_group='Purging Options', 

28 help="Comma separated list of queues names not to purge.") 

29@click.pass_context 

30@handle_preload_options 

31def purge(ctx, force, queues, exclude_queues, **kwargs): 

32 """Erase all messages from all known task queues. 

33 

34 Warning: 

35 

36 There's no undo operation for this command. 

37 """ 

38 app = ctx.obj.app 

39 queues = set(queues or app.amqp.queues.keys()) 

40 exclude_queues = set(exclude_queues or []) 

41 names = queues - exclude_queues 

42 qnum = len(names) 

43 

44 if names: 

45 queues_headline = text.pluralize(qnum, 'queue') 

46 if not force: 

47 queue_names = ', '.join(sorted(names)) 

48 click.confirm(f"{ctx.obj.style('WARNING', fg='red')}:" 

49 "This will remove all tasks from " 

50 f"{queues_headline}: {queue_names}.\n" 

51 " There is no undo for this operation!\n\n" 

52 "(to skip this prompt use the -f option)\n" 

53 "Are you sure you want to delete all tasks?", 

54 abort=True) 

55 

56 def _purge(conn, queue): 

57 try: 

58 return conn.default_channel.queue_purge(queue) or 0 

59 except conn.channel_errors: 

60 return 0 

61 

62 with app.connection_for_write() as conn: 

63 messages = sum(_purge(conn, queue) for queue in names) 

64 

65 if messages: 

66 messages_headline = text.pluralize(messages, 'message') 

67 ctx.obj.echo(f"Purged {messages} {messages_headline} from " 

68 f"{qnum} known task {queues_headline}.") 

69 else: 

70 ctx.obj.echo(f"No messages purged from {qnum} {queues_headline}.")
