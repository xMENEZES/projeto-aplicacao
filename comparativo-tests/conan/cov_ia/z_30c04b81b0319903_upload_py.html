17 def format_upload(item): 

18 if isinstance(item, dict): 

19 result = {} 

20 for k, v in item.items(): 

21 if isinstance(v, dict): 

22 v.pop("info", None) 

23 v.pop("timestamp", None) 

24 v.pop("files", None) 

25 v.pop("upload-urls", None) 

26 upload_value = v.pop("upload", None) 

27 if upload_value is not None: 

28 msg = "Uploaded" if upload_value else "Skipped, already in server" 

29 force_upload = v.pop("force_upload", None) 

30 if force_upload: 

31 msg += " - forced" 

32 k = f"{k} ({msg})"

40@conan_command(group="Creator", formatters={"text": summary_upload_list,

42def upload(conan_api: ConanAPI, parser, *args): 

43 """ 

44 Upload packages to a remote. 

45 

46 By default, all the matching references are uploaded (all revisions). 

47 By default, if a recipe reference is specified, it will upload all the revisions for all the 

48 binary packages, unless --only-recipe is specified. You can use the "latest" placeholder at the 

49 "reference" argument to specify the latest revision of the recipe or the package. 

50 """ 

51 parser.add_argument('pattern', nargs="?", 

52 help="A pattern in the form 'pkg/version#revision:package_id#revision', " 

53 "e.g: \"zlib/1.2.13:*\" means all binaries for zlib/1.2.13. " 

54 "If revision is not specified, it is assumed latest one.") 

55 parser.add_argument('-p', '--package-query', default=None, action=OnceArgument, 

56 help="Only upload packages matching a specific query. e.g: os=Windows AND " 

57 "(arch=x86 OR compiler=gcc)") 

58 # using required, we may want to pass this as a positional argument? 

59 parser.add_argument("-r", "--remote", action=OnceArgument, required=True, 

60 help='Upload to this specific remote') 

61 parser.add_argument("--only-recipe", action='store_true', default=False, 

62 help='Upload only the recipe/s, not the binary packages.') 

63 parser.add_argument("--force", action='store_true', default=False, 

64 help='Force the upload of the artifacts even if the revision already exists' 

65 ' in the server') 

66 parser.add_argument("--check", action='store_true', default=False, 

67 help='Perform an integrity check, using the manifests, before upload') 

68 parser.add_argument('-c', '--confirm', default=False, action='store_true', 

69 help='Upload all matching recipes without confirmation') 

70 parser.add_argument('--dry-run', default=False, action='store_true', 

71 help='Do not execute the real upload (experimental)') 

72 parser.add_argument('--allow-disabled', default=False, action='store_true', 

73 help='Allow uploading to disabled remote') 

74 parser.add_argument("-l", "--list", help="Package list file") 

75 parser.add_argument("-m", "--metadata", action='append', 

76 help='Upload the metadata, even if the package is already in the server and ' 

77 'not uploaded') 

78 

79 args = parser.parse_args(*args) 

80 

81 remote = conan_api.remotes.get(args.remote) 

82 enabled_remotes = conan_api.remotes.list() 

83 

84 if args.pattern is None and args.list is None: 

85 raise ConanException("Missing pattern or package list file") 

86 if args.pattern and args.list: 

87 raise ConanException("Cannot define both the pattern and the package list file") 

88 if args.package_query and args.list: 

89 raise ConanException("Cannot define package-query and the package list file") 

90 if args.allow_disabled: 

91 remote.disabled = False 

92 

93 is_prepared = False 

94 

95 if args.list: 

96 package_list, is_prepared = conan_api.upload.get_pkglist_to_upload(args.list, remote) 

97 if args.only_recipe: 

98 package_list.only_recipes() 

99 else: 

100 ref_pattern = ListPattern(args.pattern, package_id="*", only_recipe=args.only_recipe) 

101 package_list = conan_api.list.select(ref_pattern, package_query=args.package_query) 

102 

103 if package_list: 

104 # If only if search with "*" we ask for confirmation 

105 if not args.list and not args.confirm and "*" in args.pattern: 

106 package_list = _ask_confirm_upload(conan_api, package_list) 

107 

108 conan_api.upload.upload_full(package_list, remote, enabled_remotes, args.check, 

109 args.force, args.metadata, args.dry_run, 

110 is_prepared=is_prepared) 

111 else: 

112 # Don't error on no recipes for automated workflows using list, 

113 # but warn to tell the user that no packages were uploaded 

114 ConanOutput().warning("No packages were uploaded because the selection is empty.") 

115 

116 pkglist = MultiPackagesList() 

117 pkglist.add(remote.name, package_list) 

118 return {

124def _ask_confirm_upload(conan_api, package_list): 

125 ui = UserInput(conan_api.config.get("core:non_interactive")) 

126 result = PackagesList() 

127 for ref, packages in package_list.items(): 

128 msg = f"Are you sure you want to upload recipe '{ref.repr_notime()}'?" 

129 if ui.request_boolean(msg): 

130 result.add_ref(ref) 

131 ref_dict = package_list.recipe_dict(ref).copy() 

132 ref_dict.pop("packages", None) 

133 result.recipe_dict(ref).update(ref_dict) 

134 for pref, pkg_id_info in packages.items(): 

135 msg = f"Are you sure you want to upload package '{pref.repr_notime()}'?" 

136 if ui.request_boolean(msg): 

137 result.add_pref(pref, pkg_id_info) 

138 result.package_dict(pref).update(package_list.package_dict(pref)) 

139 return result
