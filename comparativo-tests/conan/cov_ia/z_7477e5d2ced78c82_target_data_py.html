209 def _get_required_components_cpp(self): 

210 """Returns a list of (component_name, DepsCppCMake)""" 

211 ret = [] 

212 sorted_comps = self.conanfile.cpp_info.get_sorted_components() 

213 pfolder_var_name = "{}_PACKAGE_FOLDER{}".format(self.pkg_name, self.config_suffix) 

214 transitive_requires = self.cmakedeps.get_transitive_requires(self.conanfile) 

215 for comp_name, comp in sorted_comps.items(): 

216 deps_cpp_cmake = _TargetDataContext(comp, pfolder_var_name, self._root_folder, 

217 self.require, self.cmake_package_type, 

218 self.is_host_windows, self.conanfile, self.cmakedeps, 

219 comp_name) 

220 

221 public_comp_deps = [] 

222 for required_pkg, required_comp in comp.parsed_requires(): 

223 if required_pkg is not None: # Points to a component of a different package 

224 try: # Make sure the declared dependency is at least in the recipe requires 

225 self.conanfile.dependencies[required_pkg] 

226 except KeyError: 

227 raise ConanException(f"{self.conanfile}: component '{comp_name}' required " 

228 f"'{required_pkg}::{required_comp}', " 

229 f"but '{required_pkg}' is not a direct dependency") 

230 try: 

231 req = transitive_requires[required_pkg] 

232 except KeyError: # The transitive dep might have been skipped 

233 pass 

234 else: 

235 public_comp_deps.append(self.get_component_alias(req, required_comp)) 

236 else: # Points to a component of same package 

237 public_comp_deps.append(self.get_component_alias(self.conanfile, required_comp)) 

238 deps_cpp_cmake.public_deps = " ".join(public_comp_deps) 

239 component_target_name = self.get_component_alias(self.conanfile, comp_name) 

240 ret.append((component_target_name, deps_cpp_cmake)) 

241 ret.reverse() 

242 return ret

256 def _get_dependencies_find_modes(self): 

257 ret = {} 

258 if self.require.build: 

259 return ret 

260 deps = self.cmakedeps.get_transitive_requires(self.conanfile) 

261 for dep in deps.values(): 

262 dep_file_name = self.cmakedeps.get_cmake_package_name(dep, self.generating_module) 

263 find_mode = self.cmakedeps.get_find_mode(dep) 

264 default_value = "NO_MODULE" if not self.generating_module else "MODULE" 

265 values = { 

266 FIND_MODE_NONE: "", 

267 FIND_MODE_CONFIG: "NO_MODULE", 

268 FIND_MODE_MODULE: "MODULE", 

269 # When the dependency is "both" or not defined, we use the one is forced 

270 # by self.find_module_mode (creating modules files-> modules, config -> config) 

271 FIND_MODE_BOTH: default_value, 

272 None: default_value} 

273 ret[dep_file_name] = values[find_mode] 

274 return ret

282 def join_paths(paths): 

283 """ 

284 Paths are doubled quoted, and escaped (but spaces) 

285 e.g: set(LIBFOO_INCLUDE_DIRS "/path/to/included/dir" "/path/to/included/dir2") 

286 """ 

287 ret = [] 

288 for p in paths: 

289 assert os.path.isabs(p), "{} is not absolute".format(p) 

290 

291 # Trying to use a ${mypkg_PACKAGE_FOLDER}/include path instead of full 

292 if p.startswith(package_folder): 

293 # Prepend the {{ pkg_name }}_PACKAGE_FOLDER{{ config_suffix }} 

294 rel = p[len(package_folder):] 

295 rel = rel.replace('\\', '/').replace('$', '\\$').replace('"', '\\"').lstrip("/") 

296 norm_path = ("${%s}/%s" % (pfolder_var_name, rel)) 

297 else: 

298 norm_path = p.replace('\\', '/').replace('$', '\\$').replace('"', '\\"') 

299 ret.append('"{}"'.format(norm_path)) 

300 

301 return "\n\t\t\t".join(ret) 

302 

303 def join_flags(separator, values): 

304 # Flags have to be escaped 

305 ret = separator.join(cmake_escape_value(v) for v in values) 

306 return ret 

307 

308 def join_defines(values, prefix=""): 

309 # Defines have to be escaped, included spaces 

310 return "\n\t\t\t".join('"%s%s"' % (prefix, cmake_escape_value(v)) for v in values)
