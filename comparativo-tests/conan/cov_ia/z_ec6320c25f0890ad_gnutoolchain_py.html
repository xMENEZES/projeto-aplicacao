208 def _resolve_compilers_mapping_variables(self): 

209 ret = {} 

210 # Configuration map 

211 compilers_mapping = {"c": "CC", "cpp": "CXX", "cuda": "NVCC", "fortran": "FC", 

212 "rc": "RC", "objc": "OBJC", "objcpp": "OBJCXX", "asm": "AS", 

213 "nm": "NM", "ranlib": "RANLIB", 

214 "objdump": "OBJDUMP", "strip": "STRIP"} 

215 # Compiler definitions by conf 

216 compilers_by_conf = self._conanfile.conf.get("tools.build:compiler_executables", 

217 default={}, check_type=dict) 

218 if compilers_by_conf: 

219 unknown = [k for k in compilers_by_conf if k not in compilers_mapping] 

220 if unknown: 

221 self._conanfile.output.warning( 

222 f"tools.build:compiler_executables: ignoring unknown key(s) " 

223 f"{sorted(unknown)}, expected one of {sorted(compilers_mapping)}", 

224 warn_tag="risk") 

225 for comp, env_var in compilers_mapping.items(): 

226 if comp in compilers_by_conf: 

227 compiler = compilers_by_conf[comp] 

228 # https://github.com/conan-io/conan/issues/13780 

229 compiler = unix_path(self._conanfile, compiler) 

230 ret[env_var] = compiler # User/tools ones have precedence 

231 return ret

234 """Initialize the default environment variables.""" 

235 # If it's an Android cross-compilation

237 if not extra_env_vars: 

238 # Normally, these are the most common default flags used by MSVC in Windows

240 extra_env_vars = {"CC": "cl -nologo", 

241 "CXX": "cl -nologo", 

242 "LD": "link -nologo", 

243 "AR": "lib", 

244 "NM": "dumpbin -symbols", 

245 "OBJDUMP": ":", 

246 "RANLIB": ":", 

247 "STRIP": ":"} 

248 # Default compilers for intel-cc 

249 else: 

250 intel_compilers = intel_cc_compilers(self._conanfile) 

251 if intel_compilers: 

252 extra_env_vars["CC"] = intel_compilers["c"] 

253 extra_env_vars["CXX"] = intel_compilers["cpp"] 

254 extra_env_vars.update(self._resolve_compilers_mapping_variables()) 

255 # Issue related: https://github.com/conan-io/conan/issues/15486 

256 if self._is_cross_building and self._conanfile.conf_build:

258 self._conanfile.conf_build.get("tools.build:compiler_executables", default={}, 

259 check_type=dict) 

260 ) 

261 if "c" in compilers_build_mapping: 

262 extra_env_vars["CC_FOR_BUILD"] = compilers_build_mapping["c"] 

263 if "cpp" in compilers_build_mapping: 

264 extra_env_vars["CXX_FOR_BUILD"] = compilers_build_mapping["cpp"]

266 for env_var, env_value in extra_env_vars.items(): 

267 self.extra_env.define(env_var, env_value)

299 def cxxflags(self): 

300 fpic = "-fPIC" if self.fpic else None 

301 ret = [self.libcxx, self.cppstd, self.arch_flag, fpic, self.msvc_runtime_flag, 

302 self.sysroot_flag] + self.threads_flags 

303 apple_flags = [self.apple_isysroot_flag, self.apple_arch_flag, self.apple_min_version_flag] 

304 apple_flags += self.apple_extra_flags

308 return self._filter_list_empty_fields(ret)

365 def _get_default_configure_install_flags(self): 

366 configure_install_flags = {"--prefix": self._prefix} 

367 # If someone want arguments but not the defaults can pass them in args manually 

368 for flag_name, cppinfo_name in [("bindir", "bindirs"), ("sbindir", "bindirs"), 

369 ("libdir", "libdirs"), ("includedir", "includedirs"), 

370 ("oldincludedir", "includedirs"), 

371 ("datarootdir", "resdirs")]: 

372 elements = getattr(self._conanfile.cpp.package, cppinfo_name) 

373 cppinfo_value = f"${{prefix}}/{elements[0]}" if elements else None 

374 if cppinfo_value: 

375 configure_install_flags[f"--{flag_name}"] = cppinfo_value 

376 return configure_install_flags

396 @property 

397 def _environment(self): 

398 env = Environment() 

399 # Flags and defines 

400 env.append("CPPFLAGS", ["-D{}".format(d) for d in self.defines]) 

401 env.append("CXXFLAGS", self.cxxflags) 

402 env.append("CFLAGS", self.cflags) 

403 env.append("LDFLAGS", self.ldflags)

409 # Objective C/C++ 

410 self._include_obj_arc_flags(env) 

411 # Let's compose with user extra env variables defined (user ones have precedence) 

412 return self.extra_env.compose_env(env)
