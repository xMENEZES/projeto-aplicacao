46@conan_subcommand() 

47def require_add(conan_api, parser, subparser, *args): 

48 """ 

49 Add a new requirement to your local conanfile as a version range. 

50 By default, it will look for the requirement versions remotely. 

51 """ 

52 subparser.add_argument("--folder", 

53 help="Path to a folder containing a recipe (conanfile.py). " 

54 "Defaults to the current directory",) 

55 subparser.add_argument("requires", nargs="*", help="Requirement name.") 

56 subparser.add_argument("-tor", "--tool", action="append", default=[], 

57 help="Tool requirement name.") 

58 subparser.add_argument("-ter", "--test", action="append", default=[], 

59 help="Test requirement name.") 

60 group = subparser.add_mutually_exclusive_group() 

61 group.add_argument("-r", "--remote", default=None, action="append", 

62 help="Remote names. Accepts wildcards ('*' means all the remotes available)") 

63 group.add_argument("-nr", "--no-remote", action="store_true", 

64 help='Do not use remote, resolve exclusively in the cache') 

65 args = parser.parse_args(*args) 

66 requires = [(r, "requires", "requirements") for r in args.requires] 

67 tool_requires = [(r, "tool_requires", "build_requirements") for r in args.tool] 

68 test_requires = [(r, "test_requires", "build_requirements") for r in args.test] 

69 if not any(requires + tool_requires + test_requires): 

70 raise ConanException("You need to add any requires, tool_requires or test_requires.") 

71 path = conan_api.local.get_conanfile_path(args.folder or ".", os.getcwd(), py=True) 

72 remotes = conan_api.remotes.list(args.remote) if not args.no_remote else [None] 

73 conanfile = load(path) 

74 ConanOutput().debug(f"Loaded conanfile from {path}.") 

75 cached_results = {} 

76 success_msgs = [] 

77 for (name, req_attr, req_func) in requires + tool_requires + test_requires: 

78 # Check if that requirement exists in the conanfile. If yes, do nothing. 

79 if re.search(rf"self\.{req_attr}\([\"']{name}", conanfile): 

80 ConanOutput().warning(f"The {req_attr} {name} is already in use.") 

81 continue 

82 if name in cached_results: 

83 # Avoid double-search in remotes/cache, e.g., protobuf 

84 reference = RecipeReference.loads(f"{name}/{cached_results[name]}") 

85 elif "/" in name: # it already brings a version 

86 reference = RecipeReference.loads(name) 

87 cached_results[name] = reference.version # caching the result 

88 else: # Search the latest version in remotes/cache 

89 ref_pattern = ListPattern(f"{name}/*") 

90 # If neither remote nor cache are defined, show results only from cache 

91 results = {} 

92 for remote in remotes: 

93 try: 

94 pkglist = conan_api.list.select(ref_pattern, remote=remote) 

95 except Exception as e: 

96 remote_name = "Cache" if remote is None else remote.name 

97 ConanOutput().warning(f"[{remote_name}] {str(e)}") 

98 else: 

99 results = pkglist.serialize() 

100 if results: 

101 break 

102 if not results: 

103 ConanOutput().error(f"Recipe {name} not found.") 

104 continue 

105 # Put the upper limit for that requirement (next major version) 

106 reference = RecipeReference.loads(results.popitem()[0]) 

107 cached_results[name] = reference.version # caching the result 

108 try: 

109 version_range = f"{reference.name}/[>={reference.version} <{str(reference.version.bump(0))}]" 

110 except ConanException: # likely cannot bump the version, using it without ranges 

111 version_range = str(reference) 

112 full_version_range = f'self.{req_attr}("{version_range}")' 

113 if full_version_range: 

114 tab_space = " " * 4 

115 if f"def {req_func}(" in conanfile: 

116 conanfile = conanfile.replace(f"def {req_func}(self):\n", 

117 f"def {req_func}(self):\n{tab_space * 2}{full_version_range}\n") 

118 else: 

119 requirements_func = f"\n{tab_space}def {req_func}(self):\n{tab_space * 2}{full_version_range}\n" 

120 conanfile += requirements_func 

121 success_msgs.append(f"Added '{version_range}' as a new {req_attr}.") 

122 save(path, conanfile) 

123 ConanOutput().success('\n'.join(success_msgs))
