69def to_meson_machine(machine_os, machine_arch): 

70 """Gets the OS system info as the Meson machine context. 

71 

72 :param machine_os: ``str`` OS name. 

73 :param machine_arch: ``str`` OS arch. 

74 :return: ``dict`` Meson machine context. 

75 """ 

76 system = _meson_system_map.get(machine_os, machine_os.lower()) 

77 default_cpu_tuple = (machine_arch.lower(), machine_arch.lower(), 'little') 

78 cpu_tuple = _meson_cpu_family_map.get(machine_arch, default_cpu_tuple) 

79 cpu_family, cpu, endian = cpu_tuple[0], cpu_tuple[1], cpu_tuple[2] 

80 context = { 

81 'system': system, 

82 'cpu_family': cpu_family, 

83 'cpu': cpu, 

84 'endian': endian, 

85 } 

86 return context

89def to_meson_value(value): 

90 """Puts any value with a valid str-like Meson format. 

91 

92 :param value: ``str``, ``bool``, or ``list``, otherwise, it will do nothing. 

93 :return: formatted value as a ``str``. 

94 """ 

95 # https://mesonbuild.com/Machine-files.html#data-types 

96 # we don't need to transform the integer values 

97 if isinstance(value, str): 

98 return f"'{value}'" 

99 elif isinstance(value, bool): 

100 return 'true' if value else 'false' 

101 elif isinstance(value, list): 

102 return '[{}]'.format(', '.join([str(to_meson_value(val)) for val in value])) 

103 elif isinstance(value, _PackageOption): 

104 ConanOutput().warning(f"Please, do not use a Conan option value directly. " 

105 f"Convert 'options.{value.name}' into a valid Python" 

106 f"data type, e.g, bool(self.options.shared)", warn_tag="deprecated") 

107 return value

110def to_cppstd_flag(conanfile, compiler, compiler_version, cppstd): 

111 """Gets a valid cppstd flag. 

112 :param conanfile: ``ConanFile`` instance. 

113 :param compiler: ``str`` compiler name. 

114 :param compiler_version: ``str`` compiler version. 

115 :param cppstd: ``str`` cppstd version. 

116 :return: ``str`` cppstd flag. 

117 """ 

118 if cppstd is None: 

119 return None 

120 if disable_flag(conanfile, "cppstd"): 

121 return None 

122 if compiler == "msvc": 

123 # Meson's logic with 'vc++X' vs 'c++X' is possibly a little outdated. 

124 # Presumably the intent is 'vc++X' is permissive and 'c++X' is not, 

125 # but '/permissive-' is the default since 16.8. 

126 flag = cppstd_msvc_flag(compiler_version, cppstd) 

127 return 'v%s' % flag if flag else None 

128 else: 

129 return f"gnu++{cppstd[3:]}" if cppstd.startswith("gnu") else f"c++{cppstd}"
