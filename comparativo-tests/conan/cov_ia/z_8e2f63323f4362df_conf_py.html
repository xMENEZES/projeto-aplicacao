294 def compose_conf_value(self, other): 

295 """ 

296 self has precedence, the "other" will add/append if possible and not conflicting, but 

297 self mandates what to do. If self has define(), without placeholder, that will remain. 

298 :type other: _ConfValue 

299 """ 

300 v_type = self._value_type 

301 o_type = other._value_type 

302 

303 important = other._important and not self._important 

304 if v_type is list and o_type is list: 

305 # If important, we swap values to prioritize the other 

306 v1, v2 = (other._value, self._value) if important else (self._value, other._value) 

307 try: 

308 index = v1.index(_ConfVarPlaceHolder) 

309 except ValueError: # It doesn't have placeholder 

310 if important: 

311 self._value = other._value 

312 else: 

313 new_value = v1[:] # do a copy 

314 new_value[index:index + 1] = v2 # replace the placeholder 

315 self._value = new_value 

316 elif v_type is dict and o_type is dict: 

317 if self._update: 

318 # only if the current one is marked as "*=" update, otherwise it remains 

319 # as this is a "compose" operation, self has priority, it is the one updating 

320 # If important, we swap values to prioritize the other 

321 v1, v2 = (other._value, self._value) if important else (self._value, other._value) 

322 new_value = v2.copy() 

323 new_value.update(v1) 

324 self._value = new_value 

325 elif important: 

326 self._value = other._value 

327 elif ((issubclass(v_type, numbers.Number) and issubclass(o_type, numbers.Number)) or 

328 # They might be different kind of numbers, so skip the check below 

329 self._value is None or other._value is None): 

330 # It means any of those values were an "unset" so doing nothing because we don't 

331 # really know the original value type 

332 if important: 

333 self._value = other._value 

334 self._value_type = other._value_type 

335 elif o_type != v_type: 

336 raise ConanException("It's not possible to compose {} values " 

337 "and {} ones.".format(v_type.__name__, o_type.__name__)) 

338 # TODO: In case of any other object types? 

339 elif important: # equal type, but just string 

340 self._value = other._value
