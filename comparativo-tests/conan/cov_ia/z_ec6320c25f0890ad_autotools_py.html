97 :ref:`tools.microsoft.unix_path() function<conan_tools_microsoft_unix_path>` 

98 :param target: (Optional, Defaulted to ``None``): Choose which target to install. 

99 :param makefile: (Optional, Defaulted to ``None``): Allow specifying a custom makefile to use instead of default "Makefile" 

100 """ 

101 if target is None: 

102 target = "install" 

103 try:

105 except ConanException: 

106 do_strip = "autotools" in self._conanfile.conf.get("tools.build:install_strip", check_type=list) 

107 if do_strip:

109 args = args if args else [] 

110 str_args = " ".join(args) 

111 if "DESTDIR=" not in str_args:
