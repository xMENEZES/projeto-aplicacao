32class Variables(OrderedDict):

35 def __init__(self):

39 def __getattribute__(self, config): 

40 try: 

41 return super(Variables, self).__getattribute__(config) 

42 except AttributeError:

45 @property 

46 def configuration_types(self): 

47 # Reverse index for the configuration_types variables 

48 ret = OrderedDict()

50 for k, v in definitions.items():
