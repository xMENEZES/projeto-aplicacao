202 def get_declared_components_targets_names(self): 

203 """Returns a list of component_name""" 

204 ret = [] 

205 sorted_comps = self.conanfile.cpp_info.get_sorted_components() 

206 for comp_name, comp in sorted_comps.items(): 

207 ret.append(self.get_component_alias(self.conanfile, comp_name)) 

208 ret.reverse() 

209 return ret

211 def get_deps_targets_names(self): 

212 """ 

213 - [{foo}::{bar}, ] of the required 

214 """ 

215 ret = [] 

216 

217 # Get a list of dependencies target names 

218 # Declared cppinfo.requires or .components[].requires 

219 transitive_reqs = self.cmakedeps.get_transitive_requires(self.conanfile) 

220 if self.conanfile.cpp_info.required_components: 

221 for dep_name, component_name in self.conanfile.cpp_info.required_components: 

222 try: 

223 # if not dep_name, it is internal, from current self.conanfile 

224 req = transitive_reqs[dep_name] if dep_name is not None else self.conanfile 

225 except KeyError: 

226 # if it raises it means the required component is not in the direct_host 

227 # dependencies, maybe it has been filtered out by traits => Skip 

228 pass 

229 else: 

230 component_name = self.get_component_alias(req, component_name) 

231 ret.append(component_name) 

232 elif transitive_reqs: 

233 # Regular external "conanfile.requires" declared, not cpp_info requires 

234 ret = [self.get_root_target_name(r) for r in transitive_reqs.values()]
