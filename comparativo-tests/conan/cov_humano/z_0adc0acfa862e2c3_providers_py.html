191 @staticmethod 

192 def _parse_error(errors, ref): 

193 """This function removes the errors array that comes from the catalog and returns a 

194 more user-friendly message if we know how to parse it, or a generic one if we don't

197 def _replace_message(message): 

198 if "not found" in message: 

199 return f"Package '{ref}' not scanned: Not found." 

200 return None 

201 

202 error_msgs = filter(bool, [_replace_message(error["message"]) for error in errors]) 

203 return next(error_msgs, "Unknown error")

208 response = self._session.post(

216 if response.status_code == 404:
