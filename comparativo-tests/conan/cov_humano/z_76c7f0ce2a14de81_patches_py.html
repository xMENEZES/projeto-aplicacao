27def patch(conanfile, base_path=None, patch_file=None, patch_string=None, strip=0, fuzz=False, **kwargs): 

28 """ 

29 Applies a diff from file (patch_file) or string (patch_string) in the conanfile.source_folder 

30 directory. The folder containing the sources can be customized with the self.folders attribute 

31 in the layout(self) method. 

32 

33 :param conanfile: the current recipe, always pass 'self' 

34 :param base_path: The path is a relative path to conanfile.export_sources_folder unless an 

35 absolute path is provided. 

36 :param patch_file: Patch file that should be applied. The path is relative to the 

37 conanfile.source_folder unless an absolute path is provided. 

38 :param patch_string: Patch string that should be applied. 

39 :param strip: Number of folders to be stripped from the path. 

40 :param fuzz: Should accept fuzzy patches. 

41 :param kwargs: Extra parameters that can be added and will contribute to output information 

42 """ 

43 

44 patch_type = kwargs.get('patch_type') or ("file" if patch_file else "string") 

45 patch_description = kwargs.get('patch_description') 

46 

47 if patch_type or patch_description: 

48 patch_type_str = ' ({})'.format(patch_type) if patch_type else '' 

49 patch_description_str = ': {}'.format(patch_description) if patch_description else '' 

50 conanfile.output.info('Apply patch{}{}'.format(patch_type_str, patch_description_str)) 

51 

52 patchlog = logging.getLogger("patch_ng") 

53 patchlog.handlers = [] 

54 patchlog.addHandler(PatchLogHandler(conanfile.output, patch_file)) 

55 

56 if patch_file: 

57 # trick *1: patch_file path could be absolute (e.g. conanfile.build_folder), in that case 

58 # the join does nothing and works. 

59 patch_path = os.path.join(conanfile.export_sources_folder, patch_file) 

60 patchset = patch_ng.fromfile(patch_path) 

61 else: 

62 patchset = patch_ng.fromstring(patch_string.encode()) 

63 

64 if not patchset: 

65 raise ConanException("Failed to parse patch: %s" % (patch_file if patch_file else "string")) 

66 

67 # trick *1 

68 root = os.path.join(conanfile.source_folder, base_path) if base_path else conanfile.source_folder 

69 if not patchset.apply(strip=strip, root=root, fuzz=fuzz): 

70 raise ConanException("Failed to apply patch: %s" % patch_file)

73def apply_conandata_patches(conanfile): 

74 """ 

75 Applies patches stored in ``conanfile.conan_data`` (read from ``conandata.yml`` file). 

76 It will apply all the patches under ``patches`` entry that matches the given 

77 ``conanfile.version``. If versions are not defined in ``conandata.yml`` it will apply all the 

78 patches directly under ``patches`` keyword. 

79 

80 The key entries will be passed as kwargs to the ``patch`` function. 

81 """ 

82 if conanfile.conan_data is None: 

83 raise ConanException("conandata.yml not defined") 

84 

85 patches = conanfile.conan_data.get('patches') 

86 if patches is None: 

87 conanfile.output.info("apply_conandata_patches(): No patches defined in conandata") 

88 return 

89 

90 if isinstance(patches, dict): 

91 assert conanfile.version, "Can only be applied if conanfile.version is already defined" 

92 entries = patches.get(str(conanfile.version), []) 

93 if entries is None: 

94 conanfile.output.warning(f"apply_conandata_patches(): No patches defined for version {conanfile.version} in conandata.yml") 

95 return 

96 elif isinstance(patches, list): 

97 entries = patches 

98 else: 

99 raise ConanException("conandata.yml 'patches' should be a list or a dict {version: list}") 

100 for it in entries: 

101 if "patch_file" in it: 

102 # The patch files are located in the root src 

103 entry = it.copy() 

104 patch_file = entry.pop("patch_file") 

105 patch_file_path = os.path.join(conanfile.export_sources_folder, patch_file) 

106 if "patch_description" not in entry: 

107 entry["patch_description"] = patch_file 

108 patch(conanfile, patch_file=patch_file_path, **entry) 

109 elif "patch_string" in it: 

110 patch(conanfile, **it) 

111 elif "patch_user" in it: 

112 pass # This will be managed directly by users, can be a command or a script execution 

113 else: 

114 raise ConanException("The 'conandata.yml' file needs a 'patch_file' or 'patch_string'" 

115 " entry for every patch to be applied")
