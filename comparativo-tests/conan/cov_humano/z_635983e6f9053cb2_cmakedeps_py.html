135 def set_property(self, dep, prop, value, build_context=False): 

136 """ 

137 Using this method you can overwrite the :ref:`property<CMakeDeps Properties>` values set by 

138 the Conan recipes from the consumer. This can be done for ``cmake_file_name``, ``cmake_target_name``, 

139 ``cmake_find_mode``, ``cmake_module_file_name``, ``cmake_module_target_name``, ``cmake_additional_variables_prefixes``, 

140 ``cmake_config_version_compat``, ``system_package_version``, 

141 ``cmake_build_modules``, ``nosoname``, ``cmake_target_aliases`` and ``cmake_extra_variables``. 

142 

143 :param dep: Name of the dependency to set the :ref:`property<CMakeDeps Properties>`. For 

144 components use the syntax: ``dep_name::component_name``. 

145 :param prop: Name of the :ref:`property<CMakeDeps Properties>`. 

146 :param value: Value of the property. Use ``None`` to invalidate any value set by the 

147 upstream recipe. 

148 :param build_context: Set to ``True`` if you want to set the property for a dependency that 

149 belongs to the build context (``False`` by default). 

150 """ 

151 build_suffix = "&build" if build_context else "" 

152 self._properties.setdefault(f"{dep}{build_suffix}", {}).update({prop: value})

154 def get_property(self, prop, dep, comp_name=None, check_type=None): 

155 dep_name = dep.ref.name 

156 build_suffix = "&build" if str( 

157 dep_name) in self.build_context_activated and dep.context == "build" else "" 

158 dep_comp = f"{str(dep_name)}::{comp_name}" if comp_name else f"{str(dep_name)}" 

159 try: 

160 value = self._properties[f"{dep_comp}{build_suffix}"][prop] 

161 if check_type is not None and not isinstance(value, check_type): 

162 raise ConanException( 

163 f'The expected type for {prop} is "{check_type.__name__}", but "{type(value).__name__}" was found') 

164 return value 

165 except KeyError: 

166 return dep.cpp_info.get_property(prop, check_type=check_type) if not comp_name \ 

167 else dep.cpp_info.components[comp_name].get_property(prop, check_type=check_type)

169 def get_cmake_package_name(self, dep, module_mode=None): 

170 """Get the name of the file for the ``find_package(XXX)`` call""" 

171 # This is used by CMakeDeps to determine: 

172 # - The filename to generate (XXX-config.cmake or FindXXX.cmake) 

173 # - The name of the defined XXX_DIR variables 

174 # - The name of transitive dependencies for calls to find_dependency 

175 if module_mode and self.get_find_mode(dep) in [FIND_MODE_MODULE, FIND_MODE_BOTH]: 

176 ret = self.get_property("cmake_module_file_name", dep) 

177 if ret: 

178 return ret 

179 ret = self.get_property("cmake_file_name", dep) 

180 return ret or dep.ref.name
