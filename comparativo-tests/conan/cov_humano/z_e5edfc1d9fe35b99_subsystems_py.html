83 env = env or []

95 msys2_mode_env.define("MSYSTEM", _msystem) 

96 msys2_mode_env.define("MSYS2_PATH_TYPE", "inherit")

104 conanfile.win_bash = wb

108 wrapped_shell = environment_wrap_command(conanfile, env, envfiles_folder, wrapped_shell, 

109 accepted_extensions=("bat", "ps1"))

117 # running without it is discouraged 

118 final_command = '{} --login -c {}'.format(wrapped_shell, wrapped_user_cmd)

122def _escape_windows_cmd(command): 

123 """ To use in a regular windows cmd.exe 

124 1. Adds escapes so the argument can be unpacked by CommandLineToArgvW() 

125 2. Adds escapes for cmd.exe so the argument survives cmd.exe's substitutions. 

126 

127 Useful to escape commands to be executed in a windows bash (msys2, cygwin etc) 

128 """ 

129 quoted_arg = cmd_args_to_string([command]) 

130 return "".join(["^%s" % arg if arg in r'()%!^"<>&|' else arg for arg in quoted_arg])

177def subsystem_path(subsystem, path):

207def get_cased_path(name): 

208 if platform.system() != "Windows": 

209 return name 

210 if not os.path.isabs(name): 

211 name = os.path.abspath(name) 

212 

213 result = [] 

214 current = name 

215 while True: 

216 parent, child = os.path.split(current) 

217 if parent == current: 

218 break 

219 

220 child_cased = child 

221 if os.path.exists(parent): 

222 children = os.listdir(parent) 

223 for c in children: 

224 if c.upper() == child.upper(): 

225 child_cased = c 

226 break 

227 result.append(child_cased) 

228 current = parent 

229 drive, _ = os.path.splitdrive(current) 

230 result.append(drive) 

231 return os.sep.join(reversed(result))
