16 def __init__(self, conanfile, folder=".", excluded=None): 

17 """ 

18 :param conanfile: Conanfile instance. 

19 :param folder: Current directory, by default ``.``, the current working directory. 

20 :param excluded: Files to be excluded from the "dirty" checks. It will compose with the 

21 configuration ``core.scm:excluded`` (the configuration has higher priority). 

22 It is a list of patterns to ``fnmatch``. 

23 """ 

24 self._conanfile = conanfile 

25 self.folder = folder 

26 self._excluded = excluded 

27 global_conf = conanfile._conan_helpers.global_conf # noqa _conan_helpers 

28 conf_excluded = global_conf.get("core.scm:excluded", check_type=list) 

29 if conf_excluded: 

30 if excluded: 

31 c = ConfDefinition() 

32 c.loads(f"core.scm:excluded={excluded}") 

33 c.update_conf_definition(global_conf) 

34 self._excluded = c.get("core.scm:excluded", check_type=list) 

35 else: 

36 self._excluded = conf_excluded 

37 self._local_url = global_conf.get("core.scm:local_url", choices=["allow", "block"])

39 def run(self, cmd, hidden_output=None): 

40 """ 

41 Executes ``git <cmd>`` 

42 

43 :return: The console output of the command. 

44 """ 

45 print_cmd = cmd if hidden_output is None else cmd.replace(hidden_output, "<hidden>") 

46 self._conanfile.output.info(f"RUN: git {print_cmd}", fg=Color.BRIGHT_BLUE) 

47 with chdir(self._conanfile, self.folder): 

48 # We tried to use self.conanfile.run(), but it didn't work: 

49 # - when using win_bash, crashing because access to .settings (forbidden in source()) 

50 # - the ``conan source`` command, not passing profiles, buildenv not injected 

51 return check_output_runner("git {}".format(cmd)).strip()

53 def get_commit(self, repository=False): 

54 """ 

55 :param repository: By default gets the commit of the defined folder, use repo=True to get 

56 the commit of the repository instead. 

57 :return: The current commit, with ``git rev-list HEAD -n 1 -- <folder>``. 

58 The latest commit is returned, irrespective of local not committed changes. 

59 """ 

60 try: 

61 # commit = self.run("rev-parse HEAD") For the whole repo 

62 # This rev-list knows to capture the last commit for the folder 

63 # --full-history is needed to not avoid wrong commits: 

64 # https://github.com/conan-io/conan/issues/10971 

65 # https://git-scm.com/docs/git-rev-list#Documentation/git-rev-list.txt-Defaultmode 

66 path = '' if repository else '-- "."' 

67 commit = self.run(f'rev-list HEAD -n 1 --full-history {path}') 

68 return commit 

69 except Exception as e: 

70 raise ConanException("Unable to get git commit in '%s': %s" % (self.folder, str(e)))

72 def get_remote_url(self, remote="origin"): 

73 """ 

74 Obtains the URL of the remote git remote repository, with ``git remote -v`` 

75 

76 **Warning!** 

77 Be aware that This method will get the output from ``git remote -v``. 

78 If you added tokens or credentials to the remote in the URL, they will be exposed. 

79 Credentials shouldn’t be added to git remotes definitions, but using a credentials manager 

80 or similar mechanism. If you still want to use this approach, it is your responsibility 

81 to strip the credentials from the result. 

82 

83 :param remote: Name of the remote git repository ('origin' by default). 

84 :return: URL of the remote git remote repository. 

85 """ 

86 remotes = self.run("remote -v") 

87 for r in remotes.splitlines(): 

88 name, url = r.split(maxsplit=1) 

89 if name == remote: 

90 url, _ = url.rsplit(None, 1) 

91 # if the url still has (fetch) or (push) at the end 

92 if url.endswith("(fetch)") or url.endswith("(push)"): 

93 url, _ = url.rsplit(None, 1) 

94 if os.path.exists(url): # Windows local directory 

95 url = url.replace("\\", "/") 

96 return url

98 def commit_in_remote(self, commit, remote="origin"): 

99 """ 

100 Checks that the given commit exists in the remote, with ``branch -r --contains <commit>`` 

101 and checking an occurrence of a branch in that remote exists. 

102 

103 :param commit: Commit to check. 

104 :param remote: Name of the remote git repository ('origin' by default). 

105 :return: True if the given commit exists in the remote, False otherwise. 

106 """ 

107 if not remote: 

108 return False 

109 # Potentially do two checks here. If the clone is a shallow clone, then we won't be 

110 # able to find the commit. 

111 try: 

112 branches = self.run("branch -r --contains {}".format(commit)) 

113 if "{}/".format(remote) in branches: 

114 return True 

115 except Exception as e: 

116 raise ConanException("Unable to check remote commit in '%s': %s" % (self.folder, str(e))) 

117 

118 try: 

119 # This will raise if commit not present. 

120 self.run(f"fetch {remote} --refetch --dry-run {commit}") 

121 return True 

122 except (Exception,): 

123 # For example, if using an older git<2.36, see 

124 # https://github.com/conan-io/conan/issues/18470, then lets try the old approach 

125 try: 

126 # This will raise if commit not present. 

127 self.run("fetch {} --dry-run --depth=1 {}".format(remote, commit)) 

128 return True 

129 except (Exception,): 

130 # Don't raise an error because it could fail for many more reasons. 

131 return False

133 def is_dirty(self, repository=False): 

134 """ 

135 Returns if the current folder is dirty, running ``git status -s`` 

136 The ``Git(..., excluded=[])`` argument and the ``core.scm:excluded`` configuration will 

137 define file patterns to be skipped from this check. 

138 

139 :param repository: By default checks if the current folder is dirty. If repository=True 

140 it will check the root repository folder instead, not the current one. 

141 :return: True, if the current folder is dirty. Otherwise, False. 

142 """ 

143 path = '' if repository else '.' 

144 status = self.run(f"status {path} --short --no-branch --untracked-files").strip() 

145 self._conanfile.output.debug(f"Git status:\n{status}") 

146 if not self._excluded: 

147 return bool(status) 

148 # Parse the status output, line by line, and match it with "_excluded" 

149 lines = [line.strip() for line in status.splitlines()] 

150 # line is of the form STATUS PATH, get the path by splitting 

151 # (Taking into account that STATUS is one word, PATH might be many) 

152 lines = [line.split(maxsplit=1)[1].strip('"') for line in lines if line] 

153 lines = [line for line in lines if not any(fnmatch.fnmatch(line, p) for p in self._excluded)] 

154 self._conanfile.output.debug(f"Filtered git status: {lines}") 

155 return bool(lines)

157 def get_url_and_commit(self, remote="origin", repository=False): 

158 """ 

159 This is an advanced method, that returns both the current commit, and the remote repository url. 

160 This method is intended to capture the current remote coordinates for a package creation, 

161 so that can be used later to build again from sources from the same commit. This is the behavior: 

162 

163 * If the repository is dirty, it will raise an exception. Doesn’t make sense to capture coordinates 

164 of something dirty, as it will not be reproducible. If there are local changes, and the 

165 user wants to test a local conan create, should commit the changes first (locally, not push the changes). 

166 

167 * If the repository is not dirty, but the commit doesn’t exist in the given remote, the method 

168 will return that commit and the URL of the local user checkout. This way, a package can be 

169 conan create created locally, testing everything works, before pushing some changes to the remote. 

170 

171 * If the repository is not dirty, and the commit exists in the specified remote, it will 

172 return that commit and the url of the remote. 

173 

174 **Warning!** 

175 Be aware that This method will get the output from ``git remote -v``. 

176 If you added tokens or credentials to the remote in the URL, they will be exposed. 

177 Credentials shouldn’t be added to git remotes definitions, but using a credentials manager 

178 or similar mechanism. If you still want to use this approach, it is your responsibility 

179 to strip the credentials from the result. 

180 

181 :param remote: Name of the remote git repository ('origin' by default). 

182 :param repository: By default gets the commit of the defined folder, use repo=True to get 

183 the commit of the repository instead. 

184 :return: (url, commit) tuple 

185 """ 

186 dirty = self.is_dirty(repository=repository) 

187 if dirty: 

188 raise ConanException("Repo is dirty, cannot capture url and commit: " 

189 "{}".format(self.folder)) 

190 commit = self.get_commit(repository=repository) 

191 url = self.get_remote_url(remote=remote) 

192 in_remote = self.commit_in_remote(commit, remote=remote) 

193 if in_remote: 

194 return url, commit 

195 if self._local_url == "block": 

196 raise ConanException(f"Current commit {commit} doesn't exist in remote {remote}\n" 

197 "Failing according to 'core.scm:local_url=block' conf") 

198 

199 if self._local_url != "allow": 

200 self._conanfile.output.warning("Current commit {} doesn't exist in remote {}\n" 

201 "This revision will not be buildable in other " 

202 "computer".format(commit, remote)) 

203 return self.get_repo_root(), commit

214 def clone(self, url, target="", args=None, hide_url=True): 

215 """ 

216 Performs a ``git clone <url> <args> <target>`` operation, where target is the target directory. 

217 

218 :param url: URL of remote repository. 

219 :param target: Target folder. 

220 :param args: Extra arguments to pass to the git clone as a list. 

221 :param hide_url: Hides the URL from the log output to prevent accidental 

222 credential leaks. Can be disabled by passing ``False``. 

223 """ 

224 args = args or [] 

225 if os.path.exists(url): 

226 url = url.replace("\\", "/") # Windows local directory 

227 mkdir(self.folder) 

228 self._conanfile.output.info("Cloning git repo") 

229 target_path = f'"{target}"' if target else "" # quote in case there are spaces in path 

230 # Avoid printing the clone command, it can contain tokens 

231 self.run('clone "{}" {} {}'.format(url, " ".join(args), target_path), 

232 hidden_output=url if hide_url else None)

234 def fetch_commit(self, url, commit, hide_url=True): 

235 """ 

236 Experimental: does a single commit fetch and checkout, instead of a full clone, 

237 should be faster. 

238 

239 :param url: URL of remote repository. 

240 :param commit: The commit ref to checkout. 

241 :param hide_url: Hides the URL from the log output to prevent accidental 

242 credential leaks. Can be disabled by passing ``False``. 

243 """ 

244 if os.path.exists(url): 

245 url = url.replace("\\", "/") # Windows local directory 

246 mkdir(self.folder) 

247 self._conanfile.output.info("Shallow fetch of git repo") 

248 self.run('init') 

249 self.run(f'remote add origin "{url}"', hidden_output=url if hide_url else None) 

250 self.run(f'fetch --depth 1 origin {commit}') 

251 self.run('checkout FETCH_HEAD')

273 def coordinates_to_conandata(self, repository=False): 

274 """ 

275 Capture the "url" and "commit" from the Git repo, calling ``get_url_and_commit()``, and then 

276 store those in the ``conandata.yml`` under the "scm" key. This information can be 

277 used later to clone and checkout the exact source point that was used to create this 

278 package, and can be useful even if the recipe uses ``exports_sources`` as mechanism to 

279 embed the sources. 

280 

281 :param repository: By default gets the commit of the defined folder, use repository=True to get 

282 the commit of the repository instead. 

283 """ 

284 scm_url, scm_commit = self.get_url_and_commit(repository=repository) 

285 update_conandata(self._conanfile, {"scm": {"commit": scm_commit, "url": scm_url}})

287 def checkout_from_conandata_coordinates(self): 

288 """ 

289 Reads the "scm" field from the ``conandata.yml``, that must contain at least "url" and 

290 "commit" and then do a ``clone(url, target=".")``, ``fetch <commit>``, followed by a ``checkout(commit)``.
