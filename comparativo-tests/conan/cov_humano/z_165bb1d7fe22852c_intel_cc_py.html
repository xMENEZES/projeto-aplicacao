1""" 

2 Intel generator module for oneAPI Toolkits. 

3 

4 For simplicity and clarity, Intel informally refers to some of the terms in this document, 

5 as listed below: 

6 

7 - ICX - Intel oneAPI DPC++/C++ Compiler 

8 - ICC Classic - Intel C++ Compiler Classic 

9 - DPCPP - Intel oneAPI DPC++/C++ Compiler 

10 

11 DPCPP is built upon ICX as the underlying C++ Compiler, therefore most of this information 

12 also applies to DPCPP. 

13 

14 Intel oneAPI Toolkit (DPC++/C++ Compiler) 

15 - Versioning: https://software.intel.com/content/www/us/en/develop/articles/oneapi-toolkit-version-to-compiler-version-mapping.html 

16 - Compiler: https://software.intel.com/content/www/us/en/develop/documentation/oneapi-dpcpp-cpp-compiler-dev-guide-and-reference/top.html 

17"""

26def _is_using_intel_oneapi(compiler_version): 

27 """Check if the Intel compiler to be used belongs to Intel oneAPI 

28 

29 Note: Intel oneAPI Toolkit first version is 2021.1 

30 """ 

31 return int(compiler_version.split(".")[0]) >= 2021

34class IntelCC: 

35 """Class that manages Intel oneAPI DPC++/C++/Classic Compilers vars generation""" 

36 

37 filename = "conanintelsetvars"

39 def __init__(self, conanfile): 

40 # Let's check the compatibility 

41 compiler_version = conanfile.settings.get_safe("compiler.version") 

42 mode = conanfile.settings.get_safe("compiler.mode") 

43 if _is_using_intel_oneapi(compiler_version): 

44 if mode != "classic" and conanfile.settings.get_safe("os") == "Darwin": 

45 raise ConanException( 

46 'macOS* is not supported for the icx/icpx or dpcpp compilers. ' 

47 'Use the "classic" mode (icc compiler) instead.') 

48 else: 

49 # Do not support legacy versions 

50 raise ConanException("You have to use 'intel' compiler which is meant for legacy " 

51 "versions like Intel Parallel Studio XE.") 

52 # Private properties 

53 self._conanfile = conanfile 

54 self._settings = conanfile.settings 

55 self._compiler_version = compiler_version 

56 self._mode = mode 

57 self._out = conanfile.output 

58 # Public properties 

59 #: arch setting 

60 self.arch = conanfile.settings.get_safe("arch")

73 def generate(self, scope=None): 

74 """Generate the Conan Intel file to be loaded in build and run environments. 

75 

76 :param scope: The scope(s) for which to generate. Can be a string or list of strings. 

77 Defaults to ["build", "run"] to enable both compilation and runtime. 

78 """ 

79 check_duplicated_generator(self, self._conanfile) 

80 intel_cc_path = self._conanfile.conf.get("tools.intel:installation_path") 

81 if intel_cc_path == "": 

82 return 

83 if intel_cc_path == '""' or intel_cc_path == "''": 

84 raise ConanException("Invalid 'tools.intel:installation_path'. To undefine it use " 

85 "an empty string, like 'tools.intel:installation_path=' without " 

86 "any quotes if used in profiles") 

87 

88 if scope is None: 

89 scope = ["build", "run"] 

90 elif isinstance(scope, str): 

91 scope = [scope] 

92 

93 if platform.system() == "Windows" and not self._conanfile.win_bash: 

94 content = textwrap.dedent("""\ 

95 @echo off 

96 {} 

97 """.format(self.command)) 

98 filename = self.filename + '.bat' 

99 else: 

100 filename = self.filename + '.sh' 

101 content = self.command 

102 from conan.tools.env.environment import create_env_script 

103 for s in scope: 

104 create_env_script(self._conanfile, content, filename, s)

117 @property 

118 def command(self): 

119 """ 

120 The Intel oneAPI DPC++/C++ Compiler includes environment configuration scripts to 

121 configure your build and development environment variables: 

122 

123 - On Linux, the file is a shell script called setvars.sh. 

124 - On Windows, the file is a batch file called setvars.bat. 

125 

126 - Linux -> ``>> . /<install-dir>/setvars.sh <arg1> <arg2> … <argn><arg1> <arg2> … <argn>`` 

127 The compiler environment script file accepts an optional target architecture 

128 argument <arg>: 

129 - intel64: Generate code and use libraries for Intel 64 architecture-based targets. 

130 - ia32: Generate code and use libraries for IA-32 architecture-based targets. 

131 

132 - Windows -> ``>> call <install-dir>\\setvars.bat [<arg1>] [<arg2>]`` 

133 Where <arg1> is optional and can be one of the following: 

134 - intel64: Generate code and use libraries for Intel 64 architecture (host and target). 

135 - ia32: Generate code and use libraries for IA-32 architecture (host and target). 

136 

137 With the dpcpp compiler, <arg1> is intel64 by default. 

138 

139 The <arg2> is optional. If specified, it is one of the following: 

140 - vs2019: Microsoft Visual Studio* 2019 

141 - vs2017: Microsoft Visual Studio 2017 

142 

143 :return: `str` setvars.sh|bat command to be run 

144 """ 

145 # Let's check if user wants to use some custom arguments to run the setvars script 

146 command_args = self._conanfile.conf.get("tools.intel:setvars_args", default="") 

147 system = platform.system() 

148 svars = "setvars.bat" if system == "Windows" else "setvars.sh" 

149 command = '"%s"' % os.path.join(self.installation_path, svars) 

150 if system == "Windows": 

151 command = "call " + command 

152 else: 

153 command = ". " + command # dot is more portable than source 

154 # If user has passed custom arguments 

155 if command_args: 

156 command += " %s" % command_args 

157 return command 

158 # Add architecture argument 

159 if self.arch == "x86_64": 

160 command += " intel64" 

161 elif self.arch == "x86": 

162 command += " ia32" 

163 else: 

164 raise ConanException("don't know how to call %s for %s" % (svars, self.arch)) 

165 

166 return command

169def intel_cc_compilers(conanfile): 

170 """ 

171 Returns default compiler executables for intel-cc based on compiler.mode setting. 

172 

173 :param conanfile: The current recipe object. 

174 :return: dict with "c" and "cpp" keys, or None if compiler is not intel-cc. 

175 """ 

176 if conanfile.settings.get_safe("compiler") != "intel-cc": 

177 return None 

178 

179 # CC/CXX explicitly set in the build environment ([buildenv]) take 

180 # precedence over the hard-coded intel-cc defaults 

181 build_env = conanfile.buildenv.vars(conanfile, scope="build") 

182 if build_env.get("CC") or build_env.get("CXX"): 

183 return None 

184 

185 mode = conanfile.settings.get_safe("compiler.mode") 

186 if mode == "classic": 

187 # The Intel C++ Compiler Classic (icc/icpc/icl) was removed from Intel oneAPI 2024.0 

188 version = conanfile.settings.get_safe("compiler.version") 

189 if version is not None and int(version.split(".")[0]) >= 2024: 

190 raise ConanException( 

191 "The Intel C++ Compiler Classic (compiler.mode=classic) was removed in Intel " 

192 f"oneAPI 2024.0, it is not available for compiler.version={version}. Use " 

193 "compiler.mode=icx (or compiler.mode=dpcpp) instead.") 

194 if conanfile.settings.get_safe("os") == "Windows": 

195 # On Windows the Intel C++ Compiler Classic driver is icl (icl.exe) for both C 

196 # and C++; icc/icpc only exist on Linux/macOS. 

197 return {"c": "icl", "cpp": "icl"} 

198 return {"c": "icc", "cpp": "icpc"} 

199 elif mode == "dpcpp": 

200 return {"c": "icx", "cpp": "dpcpp"} 

201 elif mode == "icx": 

202 if conanfile.settings.get_safe("os") == "Windows": 

203 # On Windows use the icx (icx.exe) driver, which is available across oneAPI 

204 # versions; the icx-cl (MSVC-style) driver does not exist in older versions. 

205 # Select icx-cl explicitly via tools.build:compiler_executables or [buildenv] 

206 # if the cl-compatible driver is required. 

207 return {"c": "icx", "cpp": "icx"} 

208 return {"c": "icx", "cpp": "icpx"} 

209 else: 

210 return None
