94 """Represents a single HTTP/2 Stream. 

95 

96 Stream is a bidirectional flow of bytes within an established connection,

133 # Metadata of an HTTP/2 connection stream 

134 # initialized when stream is instantiated 

135 self.metadata: dict[str, Any] = {

140 "request_sent": False,

158 # and passed to the response Deferred when completely received. 

159 "body": BytesIO(),

161 # flow control window

171 def _cancel(self, _: Any) -> None: 

172 # Close this stream as gracefully as possible 

173 # If the associated request is initiated we reset this stream 

174 # else we directly call close() method 

175 if self.metadata["request_sent"]: 

176 self.reset_stream(StreamCloseReason.CANCELLED) 

177 else: 

178 self.close(StreamCloseReason.CANCELLED)

217 def _get_request_headers(self) -> list[tuple[str, str]]: 

218 url = urlparse_cached(self._request) 

219 

220 path = url.path 

221 if url.query: 

222 path += "?" + url.query 

223 

224 # This pseudo-header field MUST NOT be empty for "http" or "https" 

225 # URIs; "http" or "https" URIs that do not contain a path component 

226 # MUST include a value of '/'. The exception to this rule is an 

227 # OPTIONS request for an "http" or "https" URI that does not include 

228 # a path component; these MUST include a ":path" pseudo-header field 

229 # with a value of '*' (refer RFC 7540 - Section 8.1.2.3) 

230 if not path: 

231 path = "*" if self._request.method == "OPTIONS" else "/" 

232 

233 # Make sure pseudo-headers comes before all the other headers 

234 headers = [ 

235 (":method", self._request.method), 

236 (":authority", url.netloc), 

237 ] 

238 

239 # The ":scheme" and ":path" pseudo-header fields MUST 

240 # be omitted for CONNECT method (refer RFC 7540 - Section 8.3) 

241 if self._request.method != "CONNECT": 

242 headers += [ 

243 (":scheme", self._protocol.metadata["uri"].scheme), 

244 (":path", path), 

245 ] 

246 

247 content_length = str(len(self._request.body)) 

248 headers.append(("Content-Length", content_length)) 

249 

250 content_length_name = self._request.headers.normkey(b"Content-Length") 

251 for name, values in self._request.headers.items(): 

252 for value_bytes in values: 

253 value = str(value_bytes, "utf-8") 

254 if name == content_length_name: 

255 if value != content_length: 

256 logger.warning( 

257 "Ignoring bad Content-Length header %r of request %r, " 

258 "sending %r instead", 

259 value, 

260 self._request, 

261 content_length, 

262 ) 

263 continue 

264 headers.append((str(name, "utf-8"), value)) 

265 

266 return headers

279 def send_data(self) -> None: 

280 """Called immediately after the headers are sent. Here we send all the 

281 data as part of the request. 

282 

283 If the content length is 0 initially then we end the stream immediately and 

284 wait for response data. 

285 """ 

286 # Firstly, check what the flow control window is for current stream. 

287 window_size = self._protocol.conn.local_flow_control_window( 

288 stream_id=self.stream_id 

289 ) 

290 

291 # Next, check what the maximum frame size is. 

292 max_frame_size = self._protocol.conn.max_outbound_frame_size 

293 

294 # We will send no more than the window size or the remaining file size 

295 # of data in this call, whichever is smaller. 

296 bytes_to_send_size = min(window_size, self.metadata["remaining_content_length"]) 

297 

298 # We now need to send a number of data frames. 

299 while bytes_to_send_size > 0: 

300 chunk_size = min(bytes_to_send_size, max_frame_size) 

301 

302 data_chunk_start_id = ( 

303 self.metadata["request_content_length"] 

304 - self.metadata["remaining_content_length"] 

305 ) 

306 data_chunk = self._request.body[ 

307 data_chunk_start_id : data_chunk_start_id + chunk_size 

308 ] 

309 

310 self._protocol.conn.send_data(self.stream_id, data_chunk, end_stream=False) 

311 

312 bytes_to_send_size -= chunk_size 

313 self.metadata["remaining_content_length"] -= chunk_size 

314 

315 self.metadata["remaining_content_length"] = max( 

316 0, self.metadata["remaining_content_length"] 

317 ) 

318 

319 # End the stream if no more data needs to be sent 

320 if self.metadata["remaining_content_length"] == 0: 

321 self._protocol.conn.end_stream(self.stream_id)

338 def receive_data(self, data: bytes, flow_controlled_length: int) -> None: 

339 self._response["body"].write(data) 

340 self._response["flow_controlled_size"] += flow_controlled_length

405 def reset_stream(self, reason: StreamCloseReason = StreamCloseReason.RESET) -> None: 

406 """Close this stream by sending a RST_FRAME to the remote peer""" 

407 # The data received so far is the body of the response built for a 

408 # stopped download, otherwise the buffer is cleared early to avoid 

409 # keeping data in memory for a long time 

410 if reason is not StreamCloseReason.STOP_DOWNLOAD: 

411 self._response["body"].truncate(0) 

412 

413 self.metadata["stream_closed_local"] = True 

414 # The remote peer may have ended the stream already, e.g. because the 

415 # whole response arrived within the data that triggered this reset, in 

416 # which case there is nothing left to reset 

417 with suppress(StreamClosedError): 

418 self._protocol.conn.reset_stream(self.stream_id, ErrorCodes.REFUSED_STREAM) 

419 self.close(reason)

421 def close( 

422 self, 

423 reason: StreamCloseReason, 

424 errors: Sequence[BaseException] | None = None, 

425 from_protocol: bool = False, 

426 ) -> None: 

427 """Based on the reason sent we will handle each case.""" 

428 # Have default value of errors as an empty list as 

429 # some cases can add a list of exceptions 

430 errors = errors or () 

431 

432 if not from_protocol: 

433 self._protocol.pop_stream(self.stream_id) 

434 

435 self.metadata["stream_closed_server"] = True 

436 

437 # We do not check for Content-Length or Transfer-Encoding in response headers 

438 # and add `partial` flag as in HTTP/1.1 as 'A request or response that includes 

439 # a payload body can include a content-length header field' (RFC 7540 - Section 8.1.2.6) 

440 

441 # NOTE: Order of handling the events is important here 

442 # As we immediately cancel the request when maxsize is exceeded while 

443 # receiving DATA_FRAME's when we have received the headers (not 

444 # having Content-Length) 

445 if reason in { 

446 StreamCloseReason.MAXSIZE_EXCEEDED, 

447 StreamCloseReason.MAXSIZE_EXCEEDED_ACTUAL, 

448 }: 

449 expected_size = int( 

450 self._response["headers"].get( 

451 b"Content-Length", self._response["flow_controlled_size"] 

452 ) 

453 ) 

454 error_msg = get_maxsize_msg( 

455 expected_size, 

456 self._download_maxsize, 

457 self._request, 

458 expected=reason == StreamCloseReason.MAXSIZE_EXCEEDED, 

459 ) 

460 logger.error(error_msg) 

461 self._deferred_response.errback(DownloadCancelledError(error_msg)) 

462 

463 elif reason in {StreamCloseReason.ENDED, StreamCloseReason.STOP_DOWNLOAD}: 

464 self._fire_response_deferred() 

465 

466 # Stream was abruptly ended here 

467 elif reason is StreamCloseReason.CANCELLED: 

468 # Client has cancelled the request. Remove all the data 

469 # received and fire the response deferred with no flags set 

470 

471 # NOTE: The data is already flushed in Stream.reset_stream() called 

472 # immediately when the stream needs to be cancelled 

473 

474 # There maybe no :status in headers, we make 

475 # HTTP Status Code: 499 - Client Closed Request 

476 if self._response["status"] is None: 

477 self._response["status"] = 499 

478 self._fire_response_deferred() 

479 

480 elif reason is StreamCloseReason.RESET: 

481 self._deferred_response.errback( 

482 ResponseFailed( 

483 [ 

484 Failure( 

485 f"Remote peer {self._protocol.metadata['ip_address']} sent RST_STREAM", 

486 ProtocolError, 

487 ) 

488 ] 

489 ) 

490 ) 

491 

492 elif reason is StreamCloseReason.CONNECTION_LOST: 

493 self._deferred_response.errback(ResponseFailed(errors)) 

494 

495 elif reason is StreamCloseReason.INACTIVE: 

496 errors = (InactiveStreamClosed(self._request), *errors) 

497 self._deferred_response.errback(ResponseFailed(errors)) 

498 

499 else: 

500 assert reason is StreamCloseReason.INVALID_HOSTNAME 

501 self._deferred_response.errback( 

502 InvalidHostname( 

503 self._request, 

504 str(self._protocol.metadata["uri"].host, "utf-8"), 

505 f"{self._protocol.metadata['ip_address']}:{self._protocol.metadata['uri'].port}", 

506 )
